lib/glass.js   import { Glass } from '../lib/glass.js'   (module singleton; no ctx; starts on first attach/watch, stops after last destroy)
  Glass.attach(target, opts) -> handle        refcounted per (target, surface); safe in enable()
    target        Clutter.Actor the glass sits behind
    opts.surface  'panel'|'menus'|'dock'|'overlays' (default 'overlays') | 'wallpaper' (plain cached blur, mode 'self')
    opts.parent   container for the layer (default target.get_parent()); layer inserted at index 0
    opts.below    optional child of parent: insert just under it instead
    opts.layers   'glass' (blur+tint+rim+highlight [+shadow], default) | 'blur' (blur only; caller keeps its tint)
    opts.mode     'backdrop' (default; own layer follows target) | 'self' (cached ACTOR blur on target itself)
    opts.follow   true (default: tracks target + ancestors) | false (caller calls handle.setRect)
    opts.radius   corner radius px of glass shape (default 0). Concentric: inner = max(outer - padding, 4)
    opts.role     'small' (default) | 'large' (tint +0.10)
    opts.shadow   bool drop shadow (default: false for panel/dock, else true)
    opts.isDark   ()=>bool, which glass colour (default: St color-scheme). Use text colour of the host.
    opts.name     actor name; opts.brightness (dark)=>number only for 'self'/'wallpaper'; opts.radius = blur px there
  handle.setVisible(bool)  layer shown iff every handle on the entry is visible and tier != 'off'
  handle.setRect(x,y,w,h)  parent coords, follow:false only
  handle.refresh()         re-resolve now (after isDark inputs change); handle.invalidate() resync geometry+repaint
  handle.destroy()         idempotent; last handle removes actors, signals, idles
  handle.tier  'off'|'blur'|'light'|'opaque'    handle.params  {tier,radius,alpha,dark,dim,motion,...} read-only
  Glass.wants(surface, rawGsettings?) -> bool   liquid-glass on and surface in liquid-glass-surfaces, OR (glass off) legacy panel-blur/dock-blur
  Glass.motion -> bool     false under reduced motion: do not animate glass
  Glass.snapshot           settings snapshot {liquid,style,blur,tint,...}
  Glass.watch(registry, fn({snapshot,env}))  now + on settings/scheme/high-contrast/reduced-motion/power-saver change
  Glass.configure(gsettings)  optional; default Extension.lookupByURL(import.meta.url).getSettings()
