lib/glass.js   import { Glass } from '../lib/glass.js'   (module singleton; no ctx needed)
  Glass.attach(target, opts) -> handle        refcounted per (target, surface); safe to call in enable()
    target   Clutter.Actor the glass sits behind
    opts.surface  'panel'|'menus'|'dock'|'overlays'|'wallpaper'   (picks settings + legacy key)
    opts.parent   container for the blur layer (default target.get_parent()); layer goes at index 0
    opts.below    optional sibling: insert just under it instead of index 0
    opts.layers   'glass' (blur+tint+highlight+shadow, default) | 'blur' (blur only; caller keeps own tint)
    opts.mode     'backdrop' (default; own layer, follows target) | 'self' (ACTOR blur on target itself)
    opts.follow   true (default; track target x/y/size) | false (caller calls handle.setRect)
    opts.radius   corner radius px of the glass shape (default 0); inner = max(outer - padding, 4) is caller math
    opts.role     'small' (default) | 'large' (tint +0.10)
    opts.shadow   bool, drop shadow (default false for panel/dock, true otherwise)
    opts.brightness  (dark:boolean)=>number, mode 'self' only
  handle.setVisible(bool)   caller hides (overview, fullscreen); layer shown iff all handles visible AND tier != 'off'
  handle.setRect(x,y,w,h)   parent coords, when follow:false
  handle.invalidate()       resync geometry + repaint now
  handle.destroy()          idempotent; last handle removes actors, signals, idle sources
  handle.tier               'off'|'blur'|'light'|'opaque'  (live)
  Glass.wants(surface)->bool        liquid-glass on and surface in liquid-glass-surfaces, OR legacy panel-blur/dock-blur on
  Glass.motion->bool                false when reduced motion; do not animate glass then
  Glass.watch(registry, fn(tier-state))  fn on any resolved change (settings, scheme, high-contrast, reduced motion, power)
  Glass.configure(gsettings)        optional; default settings come from Extension.lookupByURL
