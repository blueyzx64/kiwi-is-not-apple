lib/disposables.js  export class CleanupRegistry   (new CleanupRegistry())
  add(fn)->token                run fn on dispose
  connect(obj,sig,fn)->token    GObject signal, disconnected on dispose
  timeout(ms,fn,{name,priority}?)->token  fn returns GLib.SOURCE_CONTINUE to repeat; anything else = done, auto-forgotten. Errors logged.
  idle(fn,{name}?)->token       once
  inject(obj,name,fn)->token    obj[name]=function(...a){return fn.call(this,orig,...a)}; original restored (or own prop deleted) on dispose
  actor(a)->a                   a.destroy() on dispose; drops itself if a is destroyed first
  file(pathOrGFile)->token      deletes file on dispose. Only files Kiwi wrote.
  release(token)                run that cleanup now
  dispose()                     newest first, errors logged not thrown, safe twice, reusable. size getter = live count.

features.js  export const FEATURES = [entry...]   (registration order = start order, stop = reverse)
  ctx = { extension, settings (raw Gio.Settings), gettext, interfaceSettings }
  entry = {
    id: 'unique-string',
    setting: 'bool-key'            shorthand for when: ctx=>ctx.settings.get_boolean(key). Default when: always on.
    when: ctx=>bool                wanted?  (use for multi-key / dock conditions)
    enable: (ctx, cleanup)=>void , disable: (ctx, cleanup)=>void     OR   cls: class, built as new cls(ctx, cleanup) with .enable() / .disable()
    restartOn: ['key',..]          on that key: disable() first, then normal evaluation (re-enable if wanted)
    needs: ['id',..]               held off ('paused') unless those ids are wanted
    conflictsWith: ['id',..]       this entry WINS: those ids are disabled while this is wanted. Their saved settings untouched. Back on when winner off.
    startOnly: true                evaluated by start() only
    reapply: true                  LEGACY ONLY. enable every refresh while wanted, disable every refresh while not wanted. New features must NOT set it.
  }
  Contract: default is edge-triggered: enable() called when wanted flips on, disable() when it flips off, never twice in a row. `cleanup` is the entry's own CleanupRegistry, disposed right after disable(): register every signal/timer/actor/injection there and disable() can be empty. enable()/disable() must be sync, never async (GNOME 51). Errors are caught and logged per feature.
